const DEFAULT_FOLDER_ID = '1tm_iUo8HAdVslNVYde7LJ3sLRicavgq6';

function doGet(e) {
  const folderId = (e && e.parameter && e.parameter.folderId) || DEFAULT_FOLDER_ID;
  const folder = DriveApp.getFolderById(folderId);
  const files = folder.getFiles();
  const songs = [];
  while (files.hasNext()) {
    const file = files.next();
    const name = file.getName();
    if (/\.md$/i.test(name)) {
      songs.push({
        driveFileId: file.getId(),
        fileName: name,
        rawMarkdown: file.getBlob().getDataAsString('UTF-8'),
        updatedAt: file.getLastUpdated().getTime()
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, songs: songs }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const payload = JSON.parse((e && e.postData && e.postData.contents) || '{}');
  const folder = DriveApp.getFolderById(payload.folderId || DEFAULT_FOLDER_ID);
  const action = payload.action || 'upsert';
  const safeName = (payload.fileName || 'Untitled.md').replace(/[\/\\:*?"<>|]/g, '-');

  const existing = folder.getFilesByName(safeName);
  if (action === 'delete') {
    while (existing.hasNext()) existing.next().setTrashed(true);
    return ContentService.createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  if (existing.hasNext()) {
    const file = existing.next();
    file.setContent(payload.rawMarkdown || '');
  } else {
    folder.createFile(safeName, payload.rawMarkdown || '', MimeType.PLAIN_TEXT);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
